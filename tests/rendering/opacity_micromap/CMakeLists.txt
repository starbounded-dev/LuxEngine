cmake_minimum_required(VERSION 3.30)
project(LuxOpacityMicromapProbe LANGUAGES CXX)
find_package(Vulkan REQUIRED)
set(NRI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../Core/vendor/NRI")
set(NRI_BUILD "${NRI_ROOT}/bin/Release-windows-x86_64")
set(MESHOPT_ROOT "${NRI_ROOT}/../meshoptimizer")
add_executable(LuxOpacityMicromapProbe probe.cpp "${MESHOPT_ROOT}/opacitymap.cpp")
target_compile_features(LuxOpacityMicromapProbe PRIVATE cxx_std_20)
target_compile_definitions(LuxOpacityMicromapProbe PRIVATE NRI_STATIC_LIBRARY=1)
target_include_directories(LuxOpacityMicromapProbe PRIVATE "${NRI_ROOT}/Include" "${MESHOPT_ROOT}")
foreach(LIB NRI NRI-VK NRI-Validation NRI-NONE NRI-Shared)
  target_link_libraries(LuxOpacityMicromapProbe PRIVATE "${NRI_BUILD}/${LIB}/${LIB}.lib")
endforeach()
set(SPIRV "${CMAKE_CURRENT_BINARY_DIR}/probe.spv")
add_custom_command(OUTPUT "${SPIRV}" COMMAND "${Vulkan_GLSLC_EXECUTABLE}" --target-env=vulkan1.3 -O -fshader-stage=compute
  "${CMAKE_CURRENT_SOURCE_DIR}/probe.glsl" -o "${SPIRV}" DEPENDS probe.glsl)
add_custom_target(ProbeShader ALL DEPENDS "${SPIRV}")
add_dependencies(LuxOpacityMicromapProbe ProbeShader)
